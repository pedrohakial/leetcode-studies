146. LRU Cache (Medium)

https://leetcode.com/problems/lru-cache/ · Tópico: Design · Linked List · Hash Table

Enunciado (resumo)

Projetar um cache Least Recently Used com duas operações: get(chave) devolve o valor (ou −1) e put(chave, valor) insere/atualiza — ambas O(1) na média. Quando a capacidade estoura, o item há mais tempo sem ser usado é ejetado. Um get ou put numa chave existente conta como “uso”.

["LRUCache","put","put","get","put","get","put","get","get","get"]
[[2],[1,1],[2,2],[1],[3,3],[2],[4,4],[1],[3],[4]]
→ [null,null,null,1,null,-1,null,-1,3,4]

Padrão — o que decorar

HashMap + lista duplamente ligada: o mapa dá o acesso O(1) à chave; a lista dá a ordem de uso (mais recente na frente, LRU atrás) com remoção/ inserção O(1) em qualquer posição (por isso dupla). Toda operação segue o ritual: achou → desliga o nó da lista (remover) e religa na frente (inserir); estourou → joga fora o nó da cauda. A depender de quem pergunta, reconhecer collections.OrderedDict em Python vale menção — mas a versão didática (a da entrevista) é a lista à mão. Pista: cache com evicção por idade → a estrutura que “lembra a ordem” é a lista ligada.

Python

class NoLRU:
    __slots__ = ("chave", "valor", "ant", "prox")
    def __init__(self, chave=0, valor=0):
        self.chave = chave
        self.valor = valor
        self.ant = self.prox = None

class LRUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.mapa: dict[int, NoLRU] = {}
        self.cabeca = NoLRU()  # sentinela do lado mais recente (MRU)
        self.cauda = NoLRU()   # sentinela do lado mais antigo (LRU)
        self.cabeca.prox = self.cauda
        self.cauda.ant = self.cabeca

    def _remover(self, no: NoLRU) -> None:
        no.ant.prox = no.prox
        no.prox.ant = no.ant

    def _inserir_inicio(self, no: NoLRU) -> None:
        no.prox = self.cabeca.prox
        no.ant = self.cabeca
        self.cabeca.prox.ant = no
        self.cabeca.prox = no

    def get(self, key: int) -> int:
        no = self.mapa.get(key)
        if no is None:
            return -1
        self._remover(no)
        self._inserir_inicio(no)
        return no.valor

    def put(self, key: int, value: int) -> None:
        if key in self.mapa:
            no = self.mapa[key]
            no.valor = value
            self._remover(no)
            self._inserir_inicio(no)
            return
        if len(self.mapa) == self.cap:
            # remove o LRU (o nó real imediatamente antes da sentinela da cauda)
            lru = self.cauda.ant
            self._remover(lru)
            del self.mapa[lru.chave]
        novo = NoLRU(key, value)
        self.mapa[key] = novo
        self._inserir_inicio(novo)

Rust

#![allow(unused)]
fn main() {
use std::cell::RefCell;
use std::collections::HashMap;
use std::rc::{Rc, Weak};

struct No {
    chave: i32,
    valor: i32,
    anterior: Option<Weak<RefCell<No>>>, // Weak evita ciclo de referência
    proximo: Option<Rc<RefCell<No>>>,
}
impl No {
    fn novo(chave: i32, valor: i32) -> Rc<RefCell<No>> {
        Rc::new(RefCell::new(No { chave, valor, anterior: None, proximo: None }))
    }
}

pub struct LRUCache {
    capacidade: usize,
    mapa: HashMap<i32, Rc<RefCell<No>>>,
    // Sentinela cabeca: lado mais recente (MRU); sentinela cauda: mais antigo (LRU).
    cabeca: Rc<RefCell<No>>,
    cauda: Rc<RefCell<No>>,
}

impl LRUCache {
    pub fn new(capacity: i32) -> Self {
        let cabeca = No::novo(0, 0);
        let cauda = No::novo(0, 0);
        cabeca.borrow_mut().proximo = Some(cauda.clone());
        cauda.borrow_mut().anterior = Some(Rc::downgrade(&cabeca));
        LRUCache { capacidade: capacity as usize, mapa: HashMap::new(), cabeca, cauda }
    }

    fn remover(&self, no: &Rc<RefCell<No>>) {
        let ant = no.borrow().anterior.as_ref().and_then(Weak::upgrade).expect("nó sem anterior");
        let prox = no.borrow().proximo.clone().expect("nó sem próximo");
        ant.borrow_mut().proximo = Some(prox.clone());
        prox.borrow_mut().anterior = Some(Rc::downgrade(&ant));
    }

    fn inserir_inicio(&self, no: &Rc<RefCell<No>>) {
        let proximo_atual = self.cabeca.borrow().proximo.clone().expect("cabeca sempre tem próximo");
        no.borrow_mut().anterior = Some(Rc::downgrade(&self.cabeca));
        no.borrow_mut().proximo = Some(proximo_atual.clone());
        proximo_atual.borrow_mut().anterior = Some(Rc::downgrade(no));
        self.cabeca.borrow_mut().proximo = Some(no.clone());
    }

    pub fn get(&mut self, chave: i32) -> i32 {
        match self.mapa.get(&chave) {
            Some(no) => {
                let no = no.clone();
                self.remover(&no);
                self.inserir_inicio(&no);
                // guarda o valor num local: o Ref temporário precisa morrer
                // antes do fim do bloco (o `no` local cai junto)
                let valor = no.borrow().valor;
                valor
            }
            None => -1,
        }
    }

    pub fn put(&mut self, chave: i32, valor: i32) {
        if let Some(no) = self.mapa.get(&chave) {
            let no = no.clone();
            no.borrow_mut().valor = valor;
            self.remover(&no);
            self.inserir_inicio(&no);
            return;
        }
        if self.mapa.len() == self.capacidade {
            // LRU é o nó real imediatamente antes da sentinela da cauda
            let lru = self.cauda.borrow().anterior.as_ref().and_then(Weak::upgrade)
                .expect("cache cheio sem LRU");
            let chave_lru = lru.borrow().chave;
            self.remover(&lru);
            self.mapa.remove(&chave_lru);
        }
        let no = No::novo(chave, valor);
        self.inserir_inicio(&no);
        self.mapa.insert(chave, no);
    }
}
}

C++

class LRUCache {
    int cap;
    list<pair<int, int>> cache; // MRU na frente, LRU atrás
    unordered_map<int, list<pair<int, int>>::iterator> mapa;
public:
    LRUCache(int capacity) : cap(capacity) {}

    int get(int key) {
        auto it = mapa.find(key);
        if (it == mapa.end()) return -1;
        // splice move o nó para a frente SEM alocar (O(1))
        cache.splice(cache.begin(), cache, it->second);
        return it->second->second;
    }

    void put(int key, int value) {
        auto it = mapa.find(key);
        if (it != mapa.end()) {
            it->second->second = value; // atualiza valor
            cache.splice(cache.begin(), cache, it->second);
            return;
        }
        if ((int)cache.size() == cap) {
            auto [chave_lru, _] = cache.back(); // structured binding
            mapa.erase(chave_lru);
            cache.pop_back();
        }
        cache.emplace_front(key, value);
        mapa[key] = cache.begin();
    }
};

C

/* Simplificação documentada: arrays indexados (nós em 0..cap-1, sentinelas
 * cabeca=cap e cauda=cap+1) em vez de ponteiros — cache-friendly e sem
 * malloc/free por operação, padrão comum dos top answers em C.
 * Hash: open addressing de potência de 2 (sondagem linear, estados
 * VAZIO/ATIVO/APAGADO para a remoção não quebrar a cadeia de sondagem).
 */
typedef struct {
    int capacidade;
    int tamanho;
    int* chave; /* por slot 0..capacidade+1 (os 2 últimos são as sentinelas) */
    int* valor;
    int* ant;
    int* prox;
    unsigned hash_cap; /* potência de 2 */
    int* hash_chave;
    int* hash_slot;
    char* hash_estado; /* 0=VAZIO, 1=ATIVO, 2=APAGADO */
    int* livres;       /* pilha de slots livres */
    int topo_livres;
    int cabeca; /* índice sentinela do lado MRU */
    int cauda;  /* índice sentinela do lado LRU */
} LRUCache;

static unsigned hash_int(int chave, unsigned cap) {
    return ((unsigned)chave * 2654435761u) & (cap - 1); /* Knuth multiplicative */
}

static int achar(LRUCache* c, int chave) {
    unsigned i = hash_int(chave, c->hash_cap);
    while (c->hash_estado[i] != 0) {
        if (c->hash_estado[i] == 1 && c->hash_chave[i] == chave) return c->hash_slot[i];
        i = (i + 1) & (c->hash_cap - 1);
    }
    return -1;
}

static void inserir_hash(LRUCache* c, int chave, int slot) {
    unsigned i = hash_int(chave, c->hash_cap);
    while (c->hash_estado[i] == 1) i = (i + 1) & (c->hash_cap - 1);
    c->hash_chave[i] = chave;
    c->hash_slot[i] = slot;
    c->hash_estado[i] = 1;
}

static void remover_hash(LRUCache* c, int chave) {
    unsigned i = hash_int(chave, c->hash_cap);
    while (c->hash_estado[i] != 0) {
        if (c->hash_estado[i] == 1 && c->hash_chave[i] == chave) {
            c->hash_estado[i] = 2; /* APAGADO: tomba a cadeia sem quebrá-la */
            return;
        }
        i = (i + 1) & (c->hash_cap - 1);
    }
}

static void remover_no(LRUCache* c, int no) {
    c->prox[c->ant[no]] = c->prox[no];
    c->ant[c->prox[no]] = c->ant[no];
}

static void inserir_inicio(LRUCache* c, int no) {
    int primeiro = c->prox[c->cabeca];
    c->prox[no] = primeiro;
    c->ant[no] = c->cabeca;
    c->prox[c->cabeca] = no;
    c->ant[primeiro] = no;
}

LRUCache* lruCacheCreate(int capacity) {
    LRUCache* c = (LRUCache*)calloc(1, sizeof(LRUCache));
    int total = capacity + 2;
    c->capacidade = capacity;
    c->chave = (int*)malloc(total * sizeof(int));
    c->valor = (int*)malloc(total * sizeof(int));
    c->ant = (int*)malloc(total * sizeof(int));
    c->prox = (int*)malloc(total * sizeof(int));
    c->livres = (int*)malloc(capacity * sizeof(int));
    for (int i = 0; i < capacity; i++) c->livres[i] = capacity - 1 - i; /* topo = 0 */
    c->topo_livres = capacity;
    c->cabeca = capacity;
    c->cauda = capacity + 1;
    c->prox[c->cabeca] = c->cauda;
    c->ant[c->cauda] = c->cabeca;
    /* hash: potencia de 2 >= 4*cap (agüenta os ATIVO+APAGADO sem lotar) */
    unsigned cap = 16;
    while (cap < (unsigned)capacity * 4) cap <<= 1;
    c->hash_cap = cap;
    c->hash_chave = (int*)malloc(cap * sizeof(int));
    c->hash_slot = (int*)malloc(cap * sizeof(int));
    c->hash_estado = (char*)calloc(cap, 1);
    return c;
}

int lruCacheGet(LRUCache* obj, int key) {
    int slot = achar(obj, key);
    if (slot == -1) return -1;
    remover_no(obj, slot);
    inserir_inicio(obj, slot);
    return obj->valor[slot];
}

void lruCachePut(LRUCache* obj, int key, int value) {
    int slot = achar(obj, key);
    if (slot != -1) {
        obj->valor[slot] = value;
        remover_no(obj, slot);
        inserir_inicio(obj, slot);
        return;
    }
    if (obj->tamanho == obj->capacidade) {
        int lru = obj->ant[obj->cauda];
        remover_no(obj, lru);
        remover_hash(obj, obj->chave[lru]);
        obj->livres[obj->topo_livres++] = lru; /* devolve o slot à pilha */
        obj->tamanho--;
    }
    int novo = obj->livres[--obj->topo_livres];
    obj->chave[novo] = key;
    obj->valor[novo] = value;
    inserir_hash(obj, key, novo);
    inserir_inicio(obj, novo);
    obj->tamanho++;
}

void lruCacheFree(LRUCache* obj) {
    free(obj->chave); free(obj->valor); free(obj->ant); free(obj->prox);
    free(obj->livres); free(obj->hash_chave); free(obj->hash_slot);
    free(obj->hash_estado); free(obj);
}

Complexidade

  • Tempo: O(1) amortizado em get e put — hash O(1) + remoção/ inserção O(1) da lista dupla (o splice do C++ e os ponteiros ant/prox das demais).
  • Espaço: O(capacidade) — o mapa + os nós = no máximo capacidade entradas vivas.

Pegadinhas

  • put numa chave existente atualiza o valor E move para o “mais recente” (não conta como nova inserção, não ejeta nada).
  • As sentinelas (cabeça/cauda) simplificam: sem elas, “remover o LRU quando a lista tem 1 nó” vira um if a mais.
  • Ejeta-se o nó real antes da cauda — a cauda em si não sai da lista.
  • Em Python: o atalho oficial é collections.OrderedDict (move_to_end + popitem(last=False)) — mostre o à mão primeiro e cite o atalho.
  • Em Rust: Weak para o ponteiro de trás evita o ciclo de referência (Rc ↔ Rc nunca é dropado); borrows de RefCell devem morrer antes do fim do bloco (por isso let valor = no.borrow().valor;).
  • Em C: remoção no hash de sondagem linear precisa do estado APAGADO (senão cadeias futuras “acabam” num buraco); a simplificação de arrays indexados precisa ser dita em voz alta: “mesma lista dupla, mas com índices em vez de ponteiros”.
