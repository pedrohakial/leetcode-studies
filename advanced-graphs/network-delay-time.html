743. Network Delay Time (Medium)

https://leetcode.com/problems/network-delay-time/ · Tópico: Dijkstra (grafos ponderados)

Enunciado (resumo)

Rede de n nós (1..n) com arestas direcionadas times = [u, v, w] (w em ms). Um sinal parte do nó k e se propaga em paralelo; devolver o tempo mínimo para todos os nós receberem o sinal, ou -1 se algum ficar inalcançável.

times = [[2,1,1],[2,3,1],[3,4,1]], n = 4, k = 2  →  2   (2→1 em 1 ms; 2→3→4 em 2 ms)
times = [[1,2,1]], n = 2, k = 2                  →  -1  (a aresta entra na fonte; nó 1 nunca recebe)

Padrão — o que decorar

Dijkstra com heap de mínimos: pesos ≥ 0, e o sinal sai da fonte em todas as arestas ao mesmo tempo — exatamente “menor caminho a partir de k”. Cada aresta relaxada empurra (distância, nó) no heap; quem sai do heap com custos maior que o gravado já é entrada obsoleta (poda clássica).

Duas ideias que o enunciado pede pra verbalizar:

  1. Por que a resposta é max das distâncias, não “a última”: o nó mais lento é quem define quando a rede inteira recebeu o sinal. (Não confunda: Dijkstra relaxa distância da fonte; “a última” do raciocínio ingênuo não é a última pop do heap nem o maior id.)
  2. Heap vs O(V²): num grafo esparso (E ≪ V²), heap vence (O(E log V)); num denso (E ~ V²), a seleção linear por iteração (V iterações × O(V)) é tão boa quanto e sem o fator log — é a versão do C abaixo, que não tem heap na stdlib e nem precisa: n ≤ 100 aqui.

Python

import heapq

class Solution:
    def networkDelayTime(self, times: list[list[int]], n: int, k: int) -> int:
        grafo = [[] for _ in range(n + 1)]
        for u, v, w in times:
            grafo[u].append((v, w))

        dist = [float("inf")] * (n + 1)
        dist[k] = 0
        fila = [(0, k)]                             # (custo, nó) — min-heap
        while fila:
            custo, u = heapq.heappop(fila)
            if custo > dist[u]:
                continue                            # entrada obsoleta
            for v, w in grafo[u]:
                if custo + w < dist[v]:
                    dist[v] = custo + w
                    heapq.heappush(fila, (dist[v], v))

        atraso = max(dist[1:])
        return atraso if atraso < float("inf") else -1

Rust

#![allow(unused)]
fn main() {
use std::cmp::Reverse;
use std::collections::BinaryHeap;

impl Solution {
    pub fn network_delay_time(times: Vec<Vec<i32>>, n: i32, k: i32) -> i32 {
        let n = n as usize;
        let mut grafo = vec![vec![]; n + 1];
        for t in &times {
            let (u, v, w) = (t[0] as usize, t[1] as usize, t[2]);
            grafo[u].push((v, w));
        }

        let mut dist = vec![i32::MAX; n + 1];
        dist[k as usize] = 0;
        // BinaryHeap é MAX-heap: Reverse vira min-heap (linguagens/rust.md).
        let mut fila = BinaryHeap::new();
        fila.push(Reverse((0, k as usize)));
        while let Some(Reverse((custo, u))) = fila.pop() {
            if custo > dist[u] {
                continue; // entrada obsoleta
            }
            for &(v, w) in &grafo[u] {
                if custo + w < dist[v] {
                    dist[v] = custo + w;
                    fila.push(Reverse((dist[v], v)));
                }
            }
        }

        let atraso = dist[1..].iter().copied().max().unwrap_or(i32::MAX);
        if atraso == i32::MAX {
            -1
        } else {
            atraso
        }
    }
}
}

C++

#include <queue>
#include <vector>
#include <climits>
using namespace std;

class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
        vector<vector<pair<int, int>>> grafo(n + 1);
        for (auto& t : times) {
            grafo[t[0]].emplace_back(t[1], t[2]);
        }
        vector<int> dist(n + 1, INT_MAX);
        dist[k] = 0;
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> fila;
        fila.emplace(0, k);
        while (!fila.empty()) {
            auto [custo, u] = fila.top();
            fila.pop();
            if (custo > dist[u]) continue; // entrada obsoleta
            for (auto [v, w] : grafo[u]) {
                if (custo + w < dist[v]) {
                    dist[v] = custo + w;
                    fila.emplace(dist[v], v);
                }
            }
        }
        int atraso = 0;
        for (int i = 1; i <= n; ++i) atraso = max(atraso, dist[i]);
        return atraso == INT_MAX ? -1 : atraso;
    }
};

C

#include <stdlib.h>
#include <limits.h>

/* Dijkstra O(V²) — sem heap em C; em grafo denso (E ~ V²) isso ganha do
   heap O(E log V): seleção linear do mínimo por iteração = V×O(V) de
   seleção + V×grau_médio de relaxamento. Índice do vértice = id (1..n). */
int networkDelayTime(int** times, int timesSize, int* timesColSize,
                     int n, int k) {
    (void)timesColSize;
    const int INF = INT_MAX / 2;
    int tam = n + 1;
    int* mat = malloc((size_t)tam * tam * sizeof(int));
    int* dist = malloc((size_t)tam * sizeof(int));
    int* visitado = malloc((size_t)tam * sizeof(int));

    for (int i = 0; i < tam; i++) {
        for (int j = 0; j < tam; j++) mat[i * tam + j] = INF;
        dist[i] = INF;
        visitado[i] = 0;
    }
    for (int i = 0; i < timesSize; i++) {
        int u = times[i][0], v = times[i][1], w = times[i][2];
        if (w < mat[u * tam + v]) mat[u * tam + v] = w; // multi-aresta: fica a menor
    }
    dist[k] = 0;

    for (int passo = 0; passo < n; passo++) {
        int u = -1;
        for (int i = 1; i <= n; i++)
            if (!visitado[i] && (u == -1 || dist[i] < dist[u])) u = i;
        if (u == -1 || dist[u] == INF) break; // resto inalcançável
        visitado[u] = 1;
        for (int v = 1; v <= n; v++)
            if (!visitado[v] && mat[u * tam + v] != INF &&
                dist[u] + mat[u * tam + v] < dist[v])
                dist[v] = dist[u] + mat[u * tam + v];
    }

    int atraso = 0;
    for (int i = 1; i <= n; i++)
        if (dist[i] > atraso) atraso = dist[i];
    free(mat);
    free(dist);
    free(visitado);
    return atraso >= INF ? -1 : atraso;
}

Complexidade

  • Heap (Python/Rust/C++): tempo O(E log V) — cada aresta relaxa no máx. uma vez, cada push/pop é O(log V); espaço O(V + E) (lista de adjacência).
  • O(V²) (C): tempo O(V²) — n iterações, cada uma com seleção O(V) e relaxamento O(V); espaço O(V²) (matriz) + O(V).

Pegadinhas

  • Nós são 1-indexados: arrays de tamanho n + 1 e ignorar o índice 0.
  • Se max(dist) é infinito, algum nó não recebeu → -1 (não “0” nem o maior valor finito).
  • Arestas repetidas u→v: manter a de menor peso (na matriz, só sobrescreve se for menor — na lista de adjacência, relaxa e o min resolve).
  • Entrada obsoleta no heap (custo maior que o gravado): continue — sem ele, você só paga log a mais, mas com ele o código fica “de entrevista”.
  • C: sentinela INT_MAX / 2 para a soma dist[u] + w nunca transbordar.
