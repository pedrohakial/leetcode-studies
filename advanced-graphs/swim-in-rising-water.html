778. Swim in Rising Water (Hard)

https://leetcode.com/problems/swim-in-rising-water/ · Tópico: Dijkstra com “max no caminho” (grid)

Enunciado (resumo)

Grade n×n; no instante t a água cobre as células com grid[i][j] ≤ t. A partir de (0,0) — só navegável nas 4 direções — devolver o menor t em que dá para chegar em (n-1, n-1). O valor de cada célula é a altura que a água precisa subir para você entrar nela.

[[0,2],[1,3]]                                              →  3
[[0,1,2,3,4],[24,23,22,21,5],[12,13,14,15,16],
 [11,17,18,19,20],[10,9,8,7,6]]                            →  16

Padrão — o que decorar

Reenquadrar como caminho mínimo com uma “soma” diferente: o custo de um caminho é o máximo dos valores das células, não a soma. Queremos o mínimo sobre todos os caminhos de max(células do caminho) — e Dijkstra funciona com qualquer semigrupo monótono (soma, max, xor…): a relaxação no vizinho vira novo = max(atual, grid[ni][nj]) e o heap ordena por esse valor. A primeira vez que (n-1,n-1) sai do heap é a resposta (propriedade de Dijkstra: o pop mínimo é definitivo).

Alternativa que dá para citar: binary search no t + BFS — a “navegabilidade” é monótona (se dá no t, dá para qualquer t' > t), então for t { BFS com células ≤ t } vira binary search (0..n²) + BFS, O(n² log n) sem heap.

Python

import heapq

class Solution:
    def swimInRisingWater(self, grid: list[list[int]]) -> int:
        n = len(grid)
        melhor = [[float("inf")] * n for _ in range(n)]
        melhor[0][0] = grid[0][0]
        fila = [(grid[0][0], 0, 0)]                  # (maior até aqui, i, j)
        while fila:
            t, i, j = heapq.heappop(fila)
            if t > melhor[i][j]:
                continue                             # entrada obsoleta
            if i == n - 1 and j == n - 1:
                return t                             # 1ª pop do destino é ótima
            for di, dj in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ni, nj = i + di, j + dj
                if 0 <= ni < n and 0 <= nj < n:
                    novo = max(t, grid[ni][nj])      # custo = máx no caminho
                    if novo < melhor[ni][nj]:
                        melhor[ni][nj] = novo
                        heapq.heappush(fila, (novo, ni, nj))
        return -1                                    # inalcançável (não ocorre)

Rust

#![allow(unused)]
fn main() {
use std::cmp::Reverse;
use std::collections::BinaryHeap;

impl Solution {
    pub fn swim_in_rising_water(grid: Vec<Vec<i32>>) -> i32 {
        let n = grid.len();
        let mut melhor = vec![vec![i32::MAX; n]; n];
        melhor[0][0] = grid[0][0];
        let mut fila = BinaryHeap::new();
        fila.push(Reverse((grid[0][0], 0usize, 0usize)));
        while let Some(Reverse((t, i, j))) = fila.pop() {
            if t > melhor[i][j] {
                continue; // entrada obsoleta
            }
            if i == n - 1 && j == n - 1 {
                return t; // 1ª pop do destino é a ótima (Dijkstra)
            }
            for (di, dj) in [(1i32, 0i32), (-1, 0), (0, 1), (0, -1)] {
                let (ni, nj) = (i as i32 + di, j as i32 + dj);
                if ni < 0 || nj < 0 || ni as usize >= n || nj as usize >= n {
                    continue;
                }
                let (ni, nj) = (ni as usize, nj as usize);
                let novo = t.max(grid[ni][nj]); // custo = máx no caminho
                if novo < melhor[ni][nj] {
                    melhor[ni][nj] = novo;
                    fila.push(Reverse((novo, ni, nj)));
                }
            }
        }
        unreachable!("o problema garante caminho até (n-1, n-1)")
    }
}
}

C++

#include <vector>
#include <queue>
#include <tuple>
#include <climits>
using namespace std;

class Solution {
public:
    int swimInRisingWater(vector<vector<int>>& grid) {
        int n = grid.size();
        vector<vector<int>> melhor(n, vector<int>(n, INT_MAX));
        melhor[0][0] = grid[0][0];
        priority_queue<tuple<int, int, int>, vector<tuple<int, int, int>>,
                       greater<>>
            fila; // (maior até aqui, i, j) — min-heap
        fila.emplace(grid[0][0], 0, 0);
        int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        while (!fila.empty()) {
            auto [t, i, j] = fila.top();
            fila.pop();
            if (t > melhor[i][j]) continue; // entrada obsoleta
            if (i == n - 1 && j == n - 1) return t; // 1ª pop do destino é ótima
            for (auto& d : dirs) {
                int ni = i + d[0], nj = j + d[1];
                if (ni < 0 || nj < 0 || ni >= n || nj >= n) continue;
                int novo = max(t, grid[ni][nj]); // custo = máx no caminho
                if (novo < melhor[ni][nj]) {
                    melhor[ni][nj] = novo;
                    fila.emplace(novo, ni, nj);
                }
            }
        }
        return -1; // inalcançável (não ocorre)
    }
};

C

#include <stdlib.h>
#include <stdbool.h>
#include <limits.h>

/* Dijkstra O(V²) sem heap (C). V = n² ≤ 2500 → ~6M operações, folgado.
   O custo de uma célula é o MÁXIMO do grid no caminho até ela. */
int swimInRisingWater(int** grid, int gridSize, int* gridColSize) {
    (void)gridColSize;
    int n = gridSize;
    if (n == 1) return grid[0][0];

    int* melhor = malloc((size_t)n * n * sizeof(int));
    bool* visitado = calloc((size_t)n * n, sizeof(bool));
    for (int i = 0; i < n * n; i++) melhor[i] = INT_MAX / 2;
    melhor[0] = grid[0][0];

    for (int passo = 0; passo < n * n; passo++) {
        int alvo = -1, valor = INT_MAX / 2;
        for (int i = 0; i < n * n; i++)
            if (!visitado[i] && melhor[i] < valor) { valor = melhor[i]; alvo = i; }
        if (alvo == -1) break;
        visitado[alvo] = true;
        if (alvo == n * n - 1) {
            free(melhor);
            free(visitado);
            return valor; // 1ª seleção do destino é a ótima
        }
        int i = alvo / n, j = alvo % n;
        int viz[4][2] = {{i - 1, j}, {i + 1, j}, {i, j - 1}, {i, j + 1}};
        for (int d = 0; d < 4; d++) {
            int ni = viz[d][0], nj = viz[d][1];
            if (ni < 0 || nj < 0 || ni >= n || nj >= n) continue;
            int k = ni * n + nj;
            if (visitado[k]) continue;
            int novo = valor > grid[ni][nj] ? valor : grid[ni][nj];
            if (novo < melhor[k]) melhor[k] = novo;
        }
    }
    free(melhor);
    free(visitado);
    return -1; // inalcançável (não ocorre)
}

Complexidade

  • Heap (Python/Rust/C++): tempo O(V log V) com V = n² células — cada célula entra/sai do heap O(log V), cada aresta visitada 4 vezes; espaço O(n²) (matriz melhor + heap).
  • O(V²) (C): tempo O(V²) = O(n⁴) no pior formato — cada iteração varre todas as células para achar o mínimo. Com n ≤ 50 isso é ~6M operações, trivial. Espaço O(n²).

Pegadinhas

  • O custo é max, não soma: novo = max(t, grid[ni][nj]) — quem soma vira um problema totalmente diferente (e errado).
  • Estado do heap é (t, i, j); t é o maior valor de célula até aqui, não o tempo acumulado.
  • Grid é sempre quadrado (n×n) — não crie caso de teste com linha 1×k.
  • A 1ª pop do destino responde: parar ali é o “early exit” canônico de Dijkstra (e vale até no C, que retorna na 1ª seleção).
  • C: sem heap na stdlib; a matriz melhor guarda o min-max e visitado impede relaxar duas vezes.
