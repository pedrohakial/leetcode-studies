295. Find Median from Data Stream (Hard)

https://leetcode.com/problems/find-median-from-data-stream/ · Tópico: Heap (Priority Queue) · Design

Enunciado (resumo)

Projetar uma classe com addNum(num) e findMedian() — os números chegam um a um num fluxo; a cada momento a mediana (elemento do meio) precisa ser devolvida em O(1) (média dos dois do meio quando a contagem é par).

addNum(1); addNum(2); findMedian() → 1.5
addNum(3); findMedian() → 2.0

Padrão — o que decorar

Dois heaps balanceados: um max-heap com a metade de BAIXO e um min-heap com a metade de CIMA, invariança tam_baixo == tam_alto ou tam_baixo == tam_alto + 1. A mediana é o topo do max-heap (ímpar) ou a média dos dois topos (par). O ritual do addNum: entra no lado de baixo → move o maior dele para cima → se cima crescer demais, move o menor de cima para baixo. A diferença das MTFs é exatamente: metade esquerda ordenada descendo e metade direita subindo.

Python

class MedianFinder:
    def __init__(self):
        self.baixo = []  # max-heap (valores negados) — metade inferior
        self.alto = []   # min-heap — metade superior

    def addNum(self, num: int) -> None:
        heapq.heappush(self.baixo, -num)
        heapq.heappush(self.alto, -heapq.heappop(self.baixo))
        if len(self.alto) > len(self.baixo):
            heapq.heappush(self.baixo, -heapq.heappop(self.alto))

    def findMedian(self) -> float:
        if len(self.baixo) > len(self.alto):
            return float(-self.baixo[0])
        return (-self.baixo[0] + self.alto[0]) / 2.0

Rust

#![allow(unused)]
fn main() {
use std::cmp::Reverse;
use std::collections::BinaryHeap;

pub struct MedianFinder {
    // baixo: max-heap (metade inferior); alto: min-heap (metade superior)
    baixo: BinaryHeap<i32>,
    alto: BinaryHeap<Reverse<i32>>,
}

impl MedianFinder {
    pub fn new() -> Self {
        MedianFinder { baixo: BinaryHeap::new(), alto: BinaryHeap::new() }
    }

    pub fn add_num(&mut self, num: i32) {
        self.baixo.push(num);
        // reequilibra: o maior de baixo vai para cima
        let topo = self.baixo.pop().expect("baixo não vazio");
        self.alto.push(Reverse(topo));
        if self.alto.len() > self.baixo.len() {
            let topo = self.alto.pop().expect("alto não vazio");
            self.baixo.push(topo.0);
        }
    }

    pub fn find_median(&self) -> f64 {
        if self.baixo.len() > self.alto.len() {
            return *self.baixo.peek().unwrap() as f64; // ímpar: topo de baixo
        }
        let esquerda = *self.baixo.peek().unwrap() as f64;
        let direita = self.alto.peek().unwrap().0 as f64;
        (esquerda + direita) / 2.0 // par: média dos dois topos
    }
}
}

C++

class MedianFinder {
    priority_queue<int> baixo;                            // max-heap (inferior)
    priority_queue<int, vector<int>, greater<int>> alto;  // min-heap (superior)
public:
    MedianFinder() {}

    void addNum(int num) {
        baixo.push(num);
        alto.push(baixo.top()); // o maior de baixo sobe
        baixo.pop();
        if (alto.size() > baixo.size()) {
            baixo.push(alto.top());
            alto.pop();
        }
    }

    double findMedian() {
        if (baixo.size() > alto.size()) return baixo.top(); // ímpar
        return (baixo.top() + alto.top()) / 2.0;            // par
    }
};

C

typedef struct {
    int* baixo; /* max-heap (metade inferior) */
    int tam_baixo;
    int cap_baixo;
    int* alto; /* min-heap (metade superior) */
    int tam_alto;
    int cap_alto;
} MedianFinder;

MedianFinder* medianFinderCreate(void) {
    MedianFinder* m = (MedianFinder*)calloc(1, sizeof(MedianFinder));
    m->cap_baixo = 16;
    m->cap_alto = 16;
    m->baixo = (int*)malloc(16 * sizeof(int));
    m->alto = (int*)malloc(16 * sizeof(int));
    return m;
}

void medianFinderAddNum(MedianFinder* m, int num) {
    /* garante espaço e faz push em baixo (max-heap) */
    if (m->tam_baixo == m->cap_baixo) {
        m->cap_baixo *= 2;
        m->baixo = (int*)realloc(m->baixo, (size_t)m->cap_baixo * sizeof(int));
    }
    m->baixo[m->tam_baixo++] = num;
    subir_max(m->baixo, m->tam_baixo - 1);
    /* o maior de baixo sobe para alto */
    if (m->tam_alto == m->cap_alto) {
        m->cap_alto *= 2;
        m->alto = (int*)realloc(m->alto, (size_t)m->cap_alto * sizeof(int));
    }
    m->alto[m->tam_alto++] = m->baixo[0];
    m->baixo[0] = m->baixo[--m->tam_baixo];
    descer_max(m->baixo, m->tam_baixo, 0);
    subir_min(m->alto, m->tam_alto - 1);
    /* reequilibra: alto nunca fica maior que baixo */
    if (m->tam_alto > m->tam_baixo) {
        if (m->tam_baixo == m->cap_baixo) {
            m->cap_baixo *= 2;
            m->baixo = (int*)realloc(m->baixo, (size_t)m->cap_baixo * sizeof(int));
        }
        m->baixo[m->tam_baixo++] = m->alto[0];
        m->alto[0] = m->alto[--m->tam_alto];
        descer_min(m->alto, m->tam_alto, 0);
        subir_max(m->baixo, m->tam_baixo - 1);
    }
}

double medianFinderFindMedian(MedianFinder* m) {
    if (m->tam_baixo > m->tam_alto) return m->baixo[0]; /* ímpar */
    return (m->baixo[0] + m->alto[0]) / 2.0;            /* par */
}

Complexidade

  • addNum: O(log n) — 2 pops + 2 pushes em heaps de até n/2.
  • findMedian: O(1) — só olha os topos.
  • Espaço: O(n) — todos os elementos do fluxo (quem mantém só “um centro” com janela deslizante é outro problema).

Pegadinhas

  • A invariança precisa valer SEMPRE: se o par chega, findMedian lê os dois topos; se ímpar, lê só o de baixo — qualquer desbalanceamento quebra a resposta.
  • Elementos repetidos e negativos funcionam sem caso especial (a ordem nos heaps é por valor, e o reequilíbrio não depende da identidade); o teste clássico é um fluxo tipo [5,1,2,4,3] — mediana 3, não 4.
  • Dois topos podem somar 2·10⁵ (limite) sem estourar int — mas a média dos topos usa / 2.0 para não truncar para inteiro (só no C++/C com int a divisão por 2.0 força double).
  • Em Rust: o “max-heap de baixo” é BinaryHeap<i32> (nativo) e o “min-heap de cima” precisa de Reverse — o par de conversões é onde o sinal se inverte; peek().unwrap() é seguro porque só chamamos com o lado garantido não-vazio.
  • Em C: os dois heaps escrevem subir/descer separados (max e min) — ou se passa um comparador; arrays com realloc dobrando a capacidade para o fluxo não ter teto.
