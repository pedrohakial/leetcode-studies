208. Implement Trie (Prefix Tree) (Medium)

https://leetcode.com/problems/implement-trie-prefix-tree/ · Tópico: Design (Trie)

Enunciado (resumo)

Implementar a classe Trie com três operações sobre palavras de letras minúsculas: insert(palavra), search(palavra) (existe como PALAVRA completa) e startsWith(prefixo) (existe como prefixo de alguma palavra inserida).

Trie t;
t.insert("apple");
t.search("apple")   → true
t.search("app")     → false   (não foi inserida como palavra)
t.startsWith("app") → true
t.insert("app");
t.search("app")     → true

Padrão — o que decorar

Árvore de 26 ponteiros de filhos + flag fim. Cada nó representa um prefixo; insert desce/cria nós; search desce e exige fim == true; startsWith desce e basta existir. A raiz é um nó sentinela (não guarda caractere). O fim é o que distingue “prefixo existe” de “palavra existe” — a pegadinha central do problema. Pista: “prefixo” + “autocomplete” → trie; as 26 letras minúsculas dizem que o array de 26 filhos é melhor que um hashmap.

Python

class TrieNode:
    def __init__(self):
        self.filhos: dict[str, TrieNode] = {}
        self.fim = False

class Trie:
    def __init__(self):
        self.raiz = TrieNode()

    def insert(self, palavra: str) -> None:
        no = self.raiz
        for ch in palavra:
            no = no.filhos.setdefault(ch, TrieNode())   # cria se faltar
        no.fim = True                                   # marca fim de palavra

    def search(self, palavra: str) -> bool:
        no = self._procurar(palavra)
        return no is not None and no.fim

    def startsWith(self, prefixo: str) -> bool:
        return self._procurar(prefixo) is not None

    def _procurar(self, s: str) -> TrieNode | None:
        no = self.raiz
        for ch in s:
            if ch not in no.filhos:
                return None
            no = no.filhos[ch]
        return no

Alternativa de nó com array: filhos: list[TrieNode | None] = [None] * 26 e ord(ch) - ord('a') — mais rápido, mesmo espírito.

Rust

#![allow(unused)]
fn main() {
struct NoTrie {
    filhos: [Option<Box<NoTrie>>; 26],
    fim: bool,
}

impl NoTrie {
    fn novo() -> Self {
        NoTrie {
            // array de tamanho 26: from_fn evita o Default manual
            filhos: std::array::from_fn(|_| None),
            fim: false,
        }
    }
}

struct Trie {
    raiz: NoTrie,
}

impl Trie {
    fn new() -> Self {
        Trie { raiz: NoTrie::novo() }
    }

    fn insert(&mut self, palavra: String) {
        let mut no = &mut self.raiz;
        for b in palavra.bytes() {
            let i = (b - b'a') as usize;
            no = no.filhos[i].get_or_insert_with(|| Box::new(NoTrie::novo()));
        }
        no.fim = true;
    }

    fn search(&self, palavra: String) -> bool {
        self.procurar(&palavra).is_some_and(|no| no.fim)
    }

    fn starts_with(&self, prefixo: String) -> bool {
        self.procurar(&prefixo).is_some()
    }

    fn procurar(&self, s: &str) -> Option<&NoTrie> {
        let mut no = &self.raiz;
        for b in s.bytes() {
            no = no.filhos[(b - b'a') as usize].as_ref()?; // desce ou falha
        }
        Some(no)
    }
}
}

C++

#include <string>
using namespace std;

class Trie {
public:
    Trie() = default;

    void insert(string palavra) {
        No* no = &raiz;
        for (char ch : palavra) {
            int i = ch - 'a';
            if (!no->filhos[i]) no->filhos[i] = new No();
            no = no->filhos[i];
        }
        no->fim = true;
    }

    bool search(string palavra) {
        No* no = procurar(palavra);
        return no != nullptr && no->fim;
    }

    bool startsWith(string prefixo) {
        return procurar(prefixo) != nullptr;
    }

private:
    struct No {
        No* filhos[26] = {}; // {} zera todos os 26 ponteiros
        bool fim = false;
        ~No() {
            for (auto* f : filhos) delete f; // libera a trie inteira
        }
    };
    No raiz; // nó sentinela: a raiz não guarda caractere

    No* procurar(const string& s) {
        No* no = &raiz;
        for (char ch : s) {
            no = no->filhos[ch - 'a'];
            if (!no) return nullptr;
        }
        return no;
    }
};

C

#include <stdbool.h>
#include <stdlib.h>

typedef struct NoTrie {
    struct NoTrie* filhos[26];
    bool fim;
} NoTrie;

typedef struct {
    NoTrie raiz; /* sentinela: a raiz não guarda caractere */
} Trie;

static NoTrie* no_novo(void) {
    return (NoTrie*)calloc(1, sizeof(NoTrie)); /* zera filhos e fim */
}

Trie* trieCreate(void) {
    return (Trie*)calloc(1, sizeof(Trie));
}

static NoTrie* procurar(Trie* obj, char* s) {
    NoTrie* no = &obj->raiz;
    for (char* c = s; *c; c++) {
        int i = (unsigned char)*c - 'a';
        if (!no->filhos[i]) return NULL;
        no = no->filhos[i];
    }
    return no;
}

void trieInsert(Trie* obj, char* palavra) {
    NoTrie* no = &obj->raiz;
    for (char* c = palavra; *c; c++) {
        int i = (unsigned char)*c - 'a';
        if (!no->filhos[i]) no->filhos[i] = no_novo();
        no = no->filhos[i];
    }
    no->fim = true;
}

bool trieSearch(Trie* obj, char* palavra) {
    NoTrie* no = procurar(obj, palavra);
    return no && no->fim;
}

bool trieStartsWith(Trie* obj, char* prefixo) {
    return procurar(obj, prefixo) != NULL;
}

static void liberar_no(NoTrie* no) {
    for (int i = 0; i < 26; i++)
        if (no->filhos[i]) liberar_no(no->filhos[i]);
    free(no);
}

void trieFree(Trie* obj) {
    for (int i = 0; i < 26; i++)
        if (obj->raiz.filhos[i]) liberar_no(obj->raiz.filhos[i]);
    free(obj);
}

Complexidade

  • Tempo: O(L) por operação, L = tamanho da palavra/prefixo — desce um nível por caractere, sem busca.
  • Espaço: O(26 · L · n) no pior caso — cada prefixo distinto pode criar até 26 filhos; cada nó tem 26 ponteiros (em Python com dict é só o que é usado, em C/C++/Rust o array é fixo).

Pegadinhas

  • search ≠ startsWith: search("app") só é true se "app" foi inserida (flag fim no nó); startsWith("app") basta o prefixo existir. Esquecer o fim é o erro #1 deste problema.
  • startsWith("") é true (o nó raiz é alcançado); search("") é false a menos que a palavra vazia tenha sido inserida.
  • Entradas são somente a–z minúsculas (constraint): índice ch - 'a'. (unsigned char) no C evita i negativo se vier caractere fora do alfabeto.
  • Rust: o Box na recursão (Option<Box<NoTrie>>) é obrigatório para o tipo ser finito; std::array::from_fn(|_| None) monta os 26 slots sem Default manual.
  • C: calloc zera filhos e fim de uma vez; o trieFree precisa liberar recursivamente (o LeetCode NÃO libera a trie por você quando você cria nós com malloc).
  • Python: setdefault / dict evita if ch not in filhos; com [None]*26 não esqueça ord(ch) - 97.
