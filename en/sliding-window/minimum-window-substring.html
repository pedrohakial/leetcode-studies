76. Minimum Window Substring (Hard)

https://leetcode.com/problems/minimum-window-substring/ · Topic: Sliding Window (frequency counts)

Problem (summary)

Given strings s and t, return the smallest substring of s that contains every letter of t (counting repetitions — if t = "AABC", you need two A’s, one B, one C). If none exists, return "".

s = "ADOBECODEBANC", t = "ABC"  →  "BANC"
s = "a", t = "a"                →  "a"
s = "a", t = "aa"               →  ""   (only one 'a' available)

Pattern — what to memorize

Sliding window with a “missing” counter: count the letters of t in a map; always expand right, and for each char present in the map, decrement its count (and drop missing only when the count was still ≥ 0). When missing == 0 the window covers t — then shrink left while it stays valid, recording the best (start, size) pair at each shrink. Clue in the statement: smallest window + all letters of another string → sliding window with two maps/counts and the missing variable instead of recomparing entire maps (that’s what kills the O(26·n) factor).

Python

class Solution:
    def minWindow(self, s: str, t: str) -> str:
        if len(s) < len(t):
            return ""
        falta = len(t)
        contagens: dict[str, int] = {}
        for c in t:
            contagens[c] = contagens.get(c, 0) + 1
        esquerda = 0
        melhor = (0, len(s) + 1)  # (start, size) — sentinel size
        for direita, c in enumerate(s):
            if c in contagens:
                contagens[c] -= 1
                if contagens[c] >= 0:
                    falta -= 1
            while falta == 0:
                if direita - esquerda + 1 < melhor[1]:
                    melhor = (esquerda, direita - esquerda + 1)
                e = s[esquerda]
                if e in contagens:
                    contagens[e] += 1
                    if contagens[e] > 0:
                        falta += 1
                esquerda += 1
        inicio, tamanho = melhor
        return s[inicio:inicio + tamanho] if tamanho <= len(s) else ""

Rust

#![allow(unused)]
fn main() {
use std::collections::HashMap;

pub fn min_window(s: String, t: String) -> String {
    if s.len() < t.len() {
        return String::new();
    }
    let sb = s.as_bytes();
    let mut contagens: HashMap<u8, i32> = HashMap::new();
    for &c in t.as_bytes() {
        *contagens.entry(c).or_insert(0) += 1;
    }
    let mut falta = t.len() as i32;
    let (mut esquerda, mut melhor_inicio, mut melhor_tam) = (0, 0, usize::MAX);
    for (direita, &c) in sb.iter().enumerate() {
        if let Some(n) = contagens.get_mut(&c) {
            *n -= 1;
            if *n >= 0 {
                falta -= 1;
            }
        }
        while falta == 0 {
            let largura = direita - esquerda + 1;
            if largura < melhor_tam {
                melhor_inicio = esquerda;
                melhor_tam = largura;
            }
            let e = sb[esquerda];
            if let Some(n) = contagens.get_mut(&e) {
                *n += 1;
                if *n > 0 {
                    falta += 1;
                }
            }
            esquerda += 1;
        }
    }
    if melhor_tam == usize::MAX {
        String::new()
    } else {
        s[melhor_inicio..melhor_inicio + melhor_tam].to_string()
    }
}
}

C++

#include <string>
#include <climits>
using namespace std;

class Solution {
public:
    string minWindow(string s, string t) {
        if (s.size() < t.size()) return "";
        int contagens[128] = {};
        for (char c : t) contagens[(unsigned char)c]++;
        int falta = (int)t.size();
        int esquerda = 0, melhor_inicio = 0, melhor_tam = INT_MAX;
        for (int direita = 0; direita < (int)s.size(); ++direita) {
            int c = (unsigned char)s[direita];
            if (--contagens[c] >= 0) falta--;
            while (falta == 0) {
                int largura = direita - esquerda + 1;
                if (largura < melhor_tam) {
                    melhor_inicio = esquerda;
                    melhor_tam = largura;
                }
                int e = (unsigned char)s[esquerda];
                if (++contagens[e] > 0) falta++;
                esquerda++;
            }
        }
        return melhor_tam == INT_MAX ? "" : s.substr(melhor_inicio, melhor_tam);
    }
};

C

#include <stdlib.h>
#include <string.h>

char* minWindow(char* s, char* t) {
    int ns = strlen(s), nt = strlen(t);
    if (ns < nt) {
        char* vazio = malloc(1);
        vazio[0] = '\0';
        return vazio;
    }
    int contagens[128] = {0};
    for (int i = 0; i < nt; i++) contagens[(unsigned char)t[i]]++;
    int falta = nt;
    int esquerda = 0, melhor_inicio = 0, melhor_tam = 2147483647 /* INT_MAX */;
    for (int direita = 0; direita < ns; direita++) {
        if (--contagens[(unsigned char)s[direita]] >= 0) falta--;
        while (falta == 0) {
            int largura = direita - esquerda + 1;
            if (largura < melhor_tam) {
                melhor_inicio = esquerda;
                melhor_tam = largura;
            }
            if (++contagens[(unsigned char)s[esquerda]] > 0) falta++;
            esquerda++;
        }
    }
    if (melhor_tam == 2147483647) {
        char* vazio = malloc(1);
        vazio[0] = '\0';
        return vazio;
    }
    char* resp = malloc((melhor_tam + 1) * sizeof(char));
    memcpy(resp, s + melhor_inicio, (size_t)melhor_tam);
    resp[melhor_tam] = '\0';
    return resp;
}

Complexity

  • Time: O(n) — s is scanned twice in total (expansion + contraction; each char visits both ends at most once); map lookups are O(1) average (in C/C++ a fixed table, absolute O(1)).
  • Space: O(|t|) for the counts (C/C++: O(128) = O(1); Rust: a map with at most |t| keys).

Pitfalls

  • Repetitions count: t = "aa" requires two a’s — that’s why we decrement the count and falta only drops while the char’s count ≥ 0 (an extra a in a right step doesn’t reduce falta further).
  • Chars outside t are also scanned and ignored (they don’t touch falta), but the shrink must check the outgoing char — only restore the count if it was in the map.
  • Tied windows: either works — return the first found (compare with <, not <=).
  • In Rust, the final slice uses byte indices; the statement is ASCII (upper/lowercase), so byte = char.
  • In C/C++: without the (unsigned char) casts, high-byte chars would become negative indices into contagens[128].
