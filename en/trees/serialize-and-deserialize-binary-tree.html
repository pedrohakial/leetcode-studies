297. Serialize and Deserialize Binary Tree (Hard)

https://leetcode.com/problems/serialize-and-deserialize-binary-tree/ · Topic: Design (tree serialization)

Problem (summary)

Design two functions: serialize turns a binary tree into a string and deserialize rebuilds the SAME tree from it — the format is free but must be unambiguous (arbitrary design choice, which makes it an interview classic).

[1,2,3,null,null,4,5]  →  "1,2,#,#,3,4,#,#,5,#,#"  (this page's format)
[]                     →  "#"

Pattern — what to memorize

Preorder with a null marker (#) separated by commas and deserialization that rebuilds in the same order. The string stores the pre-order walk where every null becomes #; to deserialize, read one token at a time: # → null node, number → create the node and recurse (left, right). The format is unambiguous because preorder with nulls reconstitutes the whole tree and both sides share the order. Clue: “serialize” + tree → preorder + null marker (BFS with nulls also works, but forces you to handle "null" at the fringe).

Python

class Codec:
    def serialize(self, root):
        partes = []

        def pre(no):
            if no is None:
                partes.append("#")
                return
            partes.append(str(no.val))
            pre(no.left)
            pre(no.right)

        pre(root)
        return ",".join(partes)

    def deserialize(self, data):
        def constroi(it):
            v = next(it)
            if v == "#":
                return None
            no = TreeNode(int(v))
            no.left = constroi(it)
            no.right = constroi(it)
            return no

        return constroi(iter(data.split(",")))

Rust

#![allow(unused)]
fn main() {
use std::cell::RefCell;
use std::rc::Rc;

struct Codec {}

impl Codec {
    fn new() -> Self {
        Codec {}
    }

    fn serialize(&self, root: Option<Rc<RefCell<TreeNode>>>) -> String {
        fn pre(no: &Option<Rc<RefCell<TreeNode>>>, partes: &mut Vec<String>) {
            match no {
                None => partes.push("#".to_string()),
                Some(rc) => {
                    let no = rc.borrow();
                    partes.push(no.val.to_string());
                    pre(&no.left, partes);
                    pre(&no.right, partes);
                }
            }
        }
        let mut partes = Vec::new();
        pre(&root, &mut partes);
        partes.join(",")
    }

    fn deserialize(&self, data: String) -> Option<Rc<RefCell<TreeNode>>> {
        fn constroi(it: &mut dyn Iterator<Item = &str>) -> Option<Rc<RefCell<TreeNode>>> {
            match it.next() {
                None | Some("#") => None,
                Some(v) => {
                    let no = Rc::new(RefCell::new(TreeNode::new(v.parse().unwrap())));
                    {
                        let mut m = no.borrow_mut();
                        m.left = constroi(it);
                        m.right = constroi(it);
                    } // the RefMut dies here — safe to return the Rc
                    Some(no)
                }
            }
        }
        constroi(&mut data.split(','))
    }
}
}

C++

#include <sstream>
#include <string>
using namespace std;

class Codec {
public:
    string serialize(TreeNode* root) {
        if (!root) return "#";
        return to_string(root->val) + "," + serialize(root->left) + "," + serialize(root->right);
    }

    TreeNode* deserialize(string data) {
        istringstream iss(data);
        return constroi(iss);
    }

private:
    TreeNode* constroi(istringstream& iss) {
        string tok;
        if (!getline(iss, tok, ',')) return nullptr; // end of string
        if (tok == "#") return nullptr;
        TreeNode* no = new TreeNode(stoi(tok));
        no->left = constroi(iss);
        no->right = constroi(iss);
        return no;
    }
};

C

#include <stdio.h>
#include <stdlib.h>

/* Creates a node (LeetCode provides the struct, not the factory). */
static struct TreeNode* novo_no(int val) {
    struct TreeNode* no = (struct TreeNode*)malloc(sizeof(struct TreeNode));
    no->val = val;
    no->left = NULL;
    no->right = NULL;
    return no;
}

/* Serialize: two passes — count nodes+nulls, allocate, fill. */
static int contar(struct TreeNode* no) { /* nodes + null markers */
    if (!no) return 1;
    return 1 + contar(no->left) + contar(no->right);
}

static int preencher(struct TreeNode* no, char* saida, int pos) {
    if (!no) {
        saida[pos] = '#';
        saida[pos + 1] = ',';
        return pos + 2;
    }
    pos += sprintf(saida + pos, "%d,", no->val);
    pos = preencher(no->left, saida, pos);
    pos = preencher(no->right, saida, pos);
    return pos;
}

char* serialize(struct TreeNode* root) {
    int m = contar(root);
    /* each node: "−2147483648," (12 bytes) — extra room; null: "#," (2 bytes) */
    char* saida = (char*)malloc((size_t)m * 12 + 1);
    int pos = preencher(root, saida, 0);
    saida[pos - 1] = '\0'; /* drop the trailing comma */
    return saida;
}

/* Deserialize: index-based parser — EVERY token consumes its own separator. */
static struct TreeNode* construir(const char* d, int* p) {
    if (d[*p] == '\0' || d[*p] == ',') return NULL; /* end or loose separator */
    if (d[*p] == '#') {
        *p += 1;
        if (d[*p] == ',') *p += 1; /* the '#' also swallows its comma */
        return NULL;
    }
    int val, consumidos;
    sscanf(d + *p, "%d%n", &val, &consumidos);
    *p += consumidos;
    if (d[*p] == ',') *p += 1;

    struct TreeNode* no = novo_no(val);
    no->left = construir(d, p);
    no->right = construir(d, p);
    return no;
}

struct TreeNode* deserialize(char* data) {
    int pos = 0;
    return construir(data, &pos);
}

Complexity

  • Time: O(n) — serialize walks the tree once; deserialize reads each token once.
  • Space: O(n) — the string has ~2n tokens (# per null), + O(h) recursion in both directions.

Pitfalls

  • The preorder-with-# format is unambiguous; post/level order also work, but need the same null on both sides — the comma is mandatory in C/C++/Rust (Python’s join/split already embeds it).
  • Negative numbers: sprintf("%d,") (C) and stoi/parse handle them; the C parser needs %n to know how many characters it consumed.
  • C — the separator is each token’s own duty: a # branch that advances only 1 (leaving the comma to the next call) desyncs the recursion — advance past '#' and then its comma (see construir).
  • C++: getline(iss, tok, ',') returns false at the end of the string — the last token needs no comma; an empty iss.str() can’t happen if the string came from our serialize.
  • Rust: no.borrow_mut().left = ... followed by ...right = ... — two statements; drop the RefMut (a { } block or chained assignments) before returning the Rc.
  • Recursion vs depth: on a skewed tree recursion uses O(n) of stack — Python can blow the frame limit at n > 1000; in the interview, mention the iterative version with an explicit stack.
