295. Find Median from Data Stream (Hard)

https://leetcode.com/problems/find-median-from-data-stream/ · Topic: Heap (Priority Queue) · Design

Problem (summary)

Design a class with addNum(num) and findMedian() — the numbers arrive one at a time as a stream; at any moment the median (the middle element) must be returned in O(1) (the average of the two middle ones when the count is even).

addNum(1); addNum(2); findMedian() → 1.5
addNum(3); findMedian() → 2.0

Pattern — what to memorize

Two balanced heaps: a max-heap holding the LOWER half and a min-heap holding the UPPER half, with the invariant tam_baixo == tam_alto or tam_baixo == tam_alto + 1. The median is the max-heap’s top (odd) or the average of the two tops (even). The addNum ritual: insert into the lower side → move its largest up → if the upper side grew too big, move its smallest down. The difference from the min/max-tree variants is exactly: ordered left half going down and right half going up.

Python

class MedianFinder:
    def __init__(self):
        self.baixo = []  # max-heap (negated values) — lower half
        self.alto = []   # min-heap — upper half

    def addNum(self, num: int) -> None:
        heapq.heappush(self.baixo, -num)
        heapq.heappush(self.alto, -heapq.heappop(self.baixo))
        if len(self.alto) > len(self.baixo):
            heapq.heappush(self.baixo, -heapq.heappop(self.alto))

    def findMedian(self) -> float:
        if len(self.baixo) > len(self.alto):
            return float(-self.baixo[0])
        return (-self.baixo[0] + self.alto[0]) / 2.0

Rust

#![allow(unused)]
fn main() {
use std::cmp::Reverse;
use std::collections::BinaryHeap;

pub struct MedianFinder {
    // baixo: max-heap (lower half); alto: min-heap (upper half)
    baixo: BinaryHeap<i32>,
    alto: BinaryHeap<Reverse<i32>>,
}

impl MedianFinder {
    pub fn new() -> Self {
        MedianFinder { baixo: BinaryHeap::new(), alto: BinaryHeap::new() }
    }

    pub fn add_num(&mut self, num: i32) {
        self.baixo.push(num);
        // rebalance: the largest of the lower half moves up
        let topo = self.baixo.pop().expect("baixo not empty");
        self.alto.push(Reverse(topo));
        if self.alto.len() > self.baixo.len() {
            let topo = self.alto.pop().expect("alto not empty");
            self.baixo.push(topo.0);
        }
    }

    pub fn find_median(&self) -> f64 {
        if self.baixo.len() > self.alto.len() {
            return *self.baixo.peek().unwrap() as f64; // odd: top of the lower half
        }
        let esquerda = *self.baixo.peek().unwrap() as f64;
        let direita = self.alto.peek().unwrap().0 as f64;
        (esquerda + direita) / 2.0 // even: average of the two tops
    }
}
}

C++

class MedianFinder {
    priority_queue<int> baixo;                            // max-heap (lower)
    priority_queue<int, vector<int>, greater<int>> alto;  // min-heap (upper)
public:
    MedianFinder() {}

    void addNum(int num) {
        baixo.push(num);
        alto.push(baixo.top()); // the largest of the lower half goes up
        baixo.pop();
        if (alto.size() > baixo.size()) {
            baixo.push(alto.top());
            alto.pop();
        }
    }

    double findMedian() {
        if (baixo.size() > alto.size()) return baixo.top(); // odd
        return (baixo.top() + alto.top()) / 2.0;            // even
    }
};

C

typedef struct {
    int* baixo; /* max-heap (lower half) */
    int tam_baixo;
    int cap_baixo;
    int* alto; /* min-heap (upper half) */
    int tam_alto;
    int cap_alto;
} MedianFinder;

MedianFinder* medianFinderCreate(void) {
    MedianFinder* m = (MedianFinder*)calloc(1, sizeof(MedianFinder));
    m->cap_baixo = 16;
    m->cap_alto = 16;
    m->baixo = (int*)malloc(16 * sizeof(int));
    m->alto = (int*)malloc(16 * sizeof(int));
    return m;
}

void medianFinderAddNum(MedianFinder* m, int num) {
    /* make room and push into baixo (max-heap) */
    if (m->tam_baixo == m->cap_baixo) {
        m->cap_baixo *= 2;
        m->baixo = (int*)realloc(m->baixo, (size_t)m->cap_baixo * sizeof(int));
    }
    m->baixo[m->tam_baixo++] = num;
    subir_max(m->baixo, m->tam_baixo - 1);
    /* the largest of the lower half goes up */
    if (m->tam_alto == m->cap_alto) {
        m->cap_alto *= 2;
        m->alto = (int*)realloc(m->alto, (size_t)m->cap_alto * sizeof(int));
    }
    m->alto[m->tam_alto++] = m->baixo[0];
    m->baixo[0] = m->baixo[--m->tam_baixo];
    descer_max(m->baixo, m->tam_baixo, 0);
    subir_min(m->alto, m->tam_alto - 1);
    /* rebalance: alto never grows larger than baixo */
    if (m->tam_alto > m->tam_baixo) {
        if (m->tam_baixo == m->cap_baixo) {
            m->cap_baixo *= 2;
            m->baixo = (int*)realloc(m->baixo, (size_t)m->cap_baixo * sizeof(int));
        }
        m->baixo[m->tam_baixo++] = m->alto[0];
        m->alto[0] = m->alto[--m->tam_alto];
        descer_min(m->alto, m->tam_alto, 0);
        subir_max(m->baixo, m->tam_baixo - 1);
    }
}

double medianFinderFindMedian(MedianFinder* m) {
    if (m->tam_baixo > m->tam_alto) return m->baixo[0]; /* odd */
    return (m->baixo[0] + m->alto[0]) / 2.0;            /* even */
}

Complexity

  • addNum: O(log n) — 2 pops + 2 pushes on heaps of up to n/2.
  • findMedian: O(1) — just reads the tops.
  • Space: O(n) — every stream element (keeping only a “center” with a sliding window is a different problem).

Pitfalls

  • The invariant must ALWAYS hold: when the count is even, findMedian reads both tops; when odd, only the lower one — any imbalance breaks the answer.
  • Repeated elements and negatives work with no special case (the heap order is by value, and the rebalance doesn’t depend on identity); the classic test is a stream like [5,1,2,4,3] — median 3, not 4.
  • Two tops can sum to 2·10⁵ (the limit) without overflowing int — but the average must use / 2.0 so it doesn’t truncate to an integer (only C++/C with int need the division by 2.0 to force double).
  • In Rust: the “lower max-heap” is BinaryHeap<i32> (native) and the “upper min-heap” needs Reverse — the pair of conversions is where the sign flips; peek().unwrap() is safe because we only call it with the guaranteed non-empty side.
  • In C: the two heaps write separate subir/descer (max and min) — or pass a comparator; arrays with realloc doubling capacity so the stream has no ceiling.
