994. Rotting Oranges (Medium)

https://leetcode.com/problems/rotting-oranges/ · Topic: Multi-source BFS

Problem (summary)

An m x n grid where 0 = empty cell, 1 = fresh orange, 2 = rotten orange. Every minute, each rotten orange rots its 4-neighborhood (up, down, left, right) fresh oranges. Return the number of minutes until no fresh orange is left — or -1 if a fresh orange never rots (an isolated island).

grid = [[2,1,1],[1,1,0],[0,1,1]]  →  4
grid = [[2,1,1],[0,1,1],[1,0,1]]  →  -1   (the bottom fresh orange stays isolated)
grid = [[0,2]]                    →  0   (no fresh oranges)

Pattern — what to memorize

Multi-source BFS with layers = minutes: seed the queue with ALL rotten oranges and process one full layer per minute. Because BFS visits cells in distance-from-nearest-source order, the layer level is exactly the minute the cell rots. The frescas (fresh) counter is decremented when a cell rots — without it you’d need to rescan the grid at the end.

Statement tells: “every minute, each rotten orange rots neighbors” = uniform propagation per step ⇒ layered BFS (the inner for _ in range(len(fila)) is the layer marker). Multi-source because ALL rotten oranges start active (simultaneous filming, not sequential).

Python

class Solution:
    def orangesRotting(self, grid: list[list[int]]) -> int:
        from collections import deque
        m, n = len(grid), len(grid[0])
        fila = deque()
        frescas = 0
        for i in range(m):
            for j in range(n):
                if grid[i][j] == 2:
                    fila.append((i, j))            # ALL rotten oranges seed (multi-source)
                elif grid[i][j] == 1:
                    frescas += 1
        minutos = 0
        direcoes = ((1, 0), (-1, 0), (0, 1), (0, -1))
        while fila and frescas:                    # no fresh left => solved early
            for _ in range(len(fila)):             # ONE layer = ONE minute
                r, c = fila.popleft()
                for dr, dc in direcoes:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < m and 0 <= nc < n and grid[nr][nc] == 1:
                        grid[nr][nc] = 2
                        frescas -= 1
                        fila.append((nr, nc))
            minutos += 1
        return minutos if frescas == 0 else -1     # unreachable fresh left

Rust

#![allow(unused)]
fn main() {
impl Solution {
    pub fn oranges_rotting(grid: Vec<Vec<i32>>) -> i32 {
        use std::collections::VecDeque;
        let (m, n) = (grid.len(), grid[0].len());
        let mut grid = grid;
        let mut fila: VecDeque<(usize, usize)> = VecDeque::new();
        let mut frescas = 0;
        for i in 0..m {
            for j in 0..n {
                match grid[i][j] {
                    2 => fila.push_back((i, j)), // ALL rotten oranges seed (multi-source)
                    1 => frescas += 1,
                    _ => {}
                }
            }
        }
        let mut minutos = 0;
        let direcoes = [(1isize, 0isize), (-1, 0), (0, 1), (0, -1)];
        while !fila.is_empty() && frescas > 0 {  // no fresh left => solved early
            for _ in 0..fila.len() {             // ONE layer = ONE minute
                let (r, c) = fila.pop_front().unwrap();
                for (dr, dc) in direcoes {
                    let (nr, nc) = (r as isize + dr, c as isize + dc);
                    if nr >= 0 && nr < m as isize && nc >= 0 && nc < n as isize
                        && grid[nr as usize][nc as usize] == 1
                    {
                        grid[nr as usize][nc as usize] = 2;
                        frescas -= 1;
                        fila.push_back((nr as usize, nc as usize));
                    }
                }
            }
            minutos += 1;
        }
        if frescas == 0 { minutos } else { -1 }  // unreachable fresh left
    }
}
}

C++

#include <queue>
#include <vector>
using namespace std;

class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int m = (int)grid.size(), n = (int)grid[0].size();
        queue<pair<int, int>> fila;
        int frescas = 0;
        for (int i = 0; i < m; ++i)
            for (int j = 0; j < n; ++j) {
                if (grid[i][j] == 2) fila.push({i, j}); // ALL rotten (multi-source)
                else if (grid[i][j] == 1) ++frescas;
            }
        int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        int minutos = 0;
        while (!fila.empty() && frescas > 0) {   // no fresh left => solved early
            int camada = (int)fila.size();
            while (camada--) {                   // ONE layer = ONE minute
                auto [r, c] = fila.front();
                fila.pop();
                for (auto& d : dirs) {
                    int nr = r + d[0], nc = c + d[1];
                    if (nr >= 0 && nr < m && nc >= 0 && nc < n && grid[nr][nc] == 1) {
                        grid[nr][nc] = 2;
                        --frescas;
                        fila.push({nr, nc});
                    }
                }
            }
            ++minutos;
        }
        return frescas == 0 ? minutos : -1;      // unreachable fresh left
    }
};

C

#include <stdlib.h>

int orangesRotting(int** grid, int gridSize, int* gridColSize) {
    int cols = gridColSize[0];
    int* fila = malloc((size_t)gridSize * cols * sizeof(int)); /* each cell enters once */
    int frente = 0, tras = 0, frescas = 0;
    for (int i = 0; i < gridSize; i++)
        for (int j = 0; j < cols; j++) {
            if (grid[i][j] == 2) fila[tras++] = i * cols + j; /* ALL rotten */
            else if (grid[i][j] == 1) frescas++;
        }
    int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
    int minutos = 0;
    while (frente < tras && frescas > 0) {   /* no fresh left => solved early */
        int camada = tras - frente;
        while (camada--) {                   /* ONE layer = ONE minute */
            int pos = fila[frente++];
            int r = pos / cols, c = pos % cols;
            for (int d = 0; d < 4; d++) {
                int nr = r + dirs[d][0], nc = c + dirs[d][1];
                if (nr >= 0 && nr < gridSize && nc >= 0 && nc < cols && grid[nr][nc] == 1) {
                    grid[nr][nc] = 2;
                    frescas--;
                    fila[tras++] = nr * cols + nc;
                }
            }
        }
        minutos++;
    }
    free(fila);
    return frescas == 0 ? minutos : -1;   /* unreachable fresh left */
}

Complexity

  • Time: O(m·n) — each cell enters the queue at most once; the initial scan is O(m·n).
  • Space: O(m·n) — the queue fits the worst case when all cells are rotten; the grid is modified in place (mutates the input — OK on LeetCode).

Pitfalls

  • “No fresh oranges”: a grid with only empty/rotten cells returns 0 (the while never runs), even if there are no rotten oranges at all.
  • Impossible case: after the BFS the frescas > 0 counter ⇒ -1, not “minutes until the queue empties” (the queue drains through the circular rotten region, but the isolated one survives).
  • Don’t count the extra minute: a layer processed without rotting anything still consumes the minutos += 1 — that’s why while fila and frescas cuts early.
  • Rust: neighbor indices go through isize and come back to usize with the bounds guard — no i - 1 on usize.
  • C: the queue is an array of r*cols + c (packed positions); the grid is int** with gridColSize[0] as the column count (grid is rectangular here).
