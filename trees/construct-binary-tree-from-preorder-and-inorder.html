105. Construct Binary Tree from Preorder and Inorder (Medium)

https://leetcode.com/problems/construct-binary-tree-from-preorder-and-inorder/ · Tópico: Árvores (Divide & Conquer)

Enunciado (resumo)

Dadas as travessias preorder e inorder de uma árvore binária (valores únicos), reconstruir e devolver a raiz da árvore original.

preorder = [3,9,20,15,7], inorder = [9,3,15,20,7]
→ [3,9,20,null,null,15,7]

Padrão — o que decorar

Preorder entrega a raiz; inorder diz quem fica à esquerda e à direita. Pegue o próximo valor do preorder (raiz da faixa atual), ache sua posição k no inorder (precomputada num mapa valor→índice) e recorra: construir(esquerda do k) e construir(direita do k). O ponteiro do preorder anda só para frente; as faixas do inorder (índices) cortam a recursão. Pista: “reconstruir árvore” + duas ordens → dividir pela raiz.

Python

class Solution:
    def buildTree(self, preorder: list[int], inorder: list[int]) -> TreeNode | None:
        pos = {v: i for i, v in enumerate(inorder)}   # valor → índice, O(1)
        self.pre_i = 0

        def constroi(ini: int, fim: int) -> TreeNode | None:
            if ini > fim:
                return None
            val = preorder[self.pre_i]      # próxima raiz do preorder
            self.pre_i += 1
            no = TreeNode(val)
            k = pos[val]                    # tudo antes de k fica à esquerda
            no.left = constroi(ini, k - 1)
            no.right = constroi(k + 1, fim)
            return no

        return constroi(0, len(inorder) - 1)

Rust

#![allow(unused)]
fn main() {
use std::cell::RefCell;
use std::collections::HashMap;
use std::rc::Rc;

impl Solution {
    pub fn build_tree(preorder: Vec<i32>, inorder: Vec<i32>) -> Option<Rc<RefCell<TreeNode>>> {
        // Faixas em i32: `k - 1` com k = 0 vira -1 → ini(0) > fim(-1) → nulo,
        // sem underflow de usize.
        fn constroi(
            pre: &[i32],
            pre_i: &mut usize,
            pos: &HashMap<i32, usize>,
            ini: i32,
            fim: i32,
        ) -> Option<Rc<RefCell<TreeNode>>> {
            if ini > fim {
                return None;
            }
            let val = pre[*pre_i];
            *pre_i += 1; // preorder entrega a raiz de cada faixa em ordem
            let no = Rc::new(RefCell::new(TreeNode::new(val)));
            let k = pos[&val] as i32;
            no.borrow_mut().left = constroi(pre, pre_i, pos, ini, k - 1);
            no.borrow_mut().right = constroi(pre, pre_i, pos, k + 1, fim);
            Some(no)
        }

        if inorder.is_empty() {
            return None;
        }
        let pos: HashMap<i32, usize> = inorder.iter().enumerate().map(|(i, &v)| (v, i)).collect();
        let mut pre_i = 0;
        constroi(&preorder, &mut pre_i, &pos, 0, inorder.len() as i32 - 1)
    }
}
}

C++

#include <unordered_map>
#include <vector>
using namespace std;

class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        int n = (int)inorder.size();
        pos.clear();
        for (int i = 0; i < n; i++) pos[inorder[i]] = i;
        pre_i = 0;
        return constroi(preorder, 0, n - 1);
    }

private:
    unordered_map<int, int> pos; // valor → índice no inorder (valores únicos)
    int pre_i = 0;

    TreeNode* constroi(vector<int>& pre, int ini, int fim) {
        if (ini > fim) return nullptr;
        int val = pre[pre_i++]; // próxima da ordem preorder = raiz da faixa
        TreeNode* no = new TreeNode(val);
        int k = pos[val]; // tudo antes de k em inorder fica à esquerda
        no->left = constroi(pre, ini, k - 1);
        no->right = constroi(pre, k + 1, fim);
        return no;
    }
};

C

#include <stdlib.h>

/* Mapa valor → índice em open addressing (padrão da casa, ver two-sum). */
typedef struct {
    int chave;
    int indice;
    int usado;
} Entrada;

static unsigned hash_chave(int chave, unsigned mask) {
    return ((unsigned)chave * 2654435761u) & mask; /* Knuth multiplicative */
}

static int procurar(Entrada* pos, unsigned mask, int chave) {
    unsigned h = hash_chave(chave, mask);
    while (pos[h].usado) {
        if (pos[h].chave == chave) return pos[h].indice;
        h = (h + 1) & mask; /* sondagem linear */
    }
    return -1; /* não ocorre: valores são únicos */
}

/* Cria um nó de árvore (o LeetCode fornece a struct, não a fábrica). */
static struct TreeNode* novo_no(int val) {
    struct TreeNode* no = (struct TreeNode*)malloc(sizeof(struct TreeNode));
    no->val = val;
    no->left = NULL;
    no->right = NULL;
    return no;
}

static struct TreeNode* constroi(int* pre, int* pre_i, Entrada* pos, unsigned mask, int ini,
                                 int fim) {
    if (ini > fim) return NULL;
    int val = pre[(*pre_i)++]; /* próxima do preorder = raiz da faixa */
    struct TreeNode* no = novo_no(val);
    int k = procurar(pos, mask, val);
    no->left = constroi(pre, pre_i, pos, mask, ini, k - 1);
    no->right = constroi(pre, pre_i, pos, mask, k + 1, fim);
    return no;
}

struct TreeNode* buildTree(int* preorder, int preorderSize, int* inorder, int inorderSize) {
    (void)preorderSize;
    if (inorderSize == 0) return NULL;

    unsigned cap = 16;
    while (cap < (unsigned)inorderSize * 2u) cap <<= 1; /* load factor < 0.5 */
    Entrada* pos = (Entrada*)calloc(cap, sizeof(Entrada));
    unsigned mask = cap - 1;
    for (int i = 0; i < inorderSize; i++) {
        unsigned h = hash_chave(inorder[i], mask);
        while (pos[h].usado) h = (h + 1) & mask;
        pos[h].chave = inorder[i];
        pos[h].indice = i;
        pos[h].usado = 1;
    }

    int pre_i = 0;
    struct TreeNode* raiz = constroi(preorder, &pre_i, pos, mask, 0, inorderSize - 1);
    free(pos);
    return raiz;
}

Complexidade

  • Tempo: O(n) — cada valor do preorder é lido uma vez e cada k é uma busca O(1) no mapa (C com hash de open addressing).
  • Espaço: O(n) do mapa + O(h) da recursão (pior caso O(n) na enviesada).

Pegadinhas

  • Valores únicos são o contrato: sem isso o mapa valor→índice quebra.
  • O ponteiro do preorder anda sempre para frente (mesmo nas chamadas de subárvore esquerda) — um pre_i por referência (ou membro da classe) é obrigatório; cada chamada cria novo_no com o próximo valor.
  • Faixas do inorder: k-1 no lado esquerdo — com k = 0 vira -1; em Rust use i32 para as faixas (usize faria underflow); em C/C++ int já cobre.
  • C: o preorderSize é ignorado (o tamanho real vem do inorder) — o compilador avisa parâmetro não usado; (void)preorderSize silencia.
  • C: valores negativos funcionam no hash ((unsigned)chave antes da multiplicação), como no two-sum.
  • Postorder + inorder é o mesmo padrão, só que pre_i vira um ponteiro que anda do FIM para o início.
